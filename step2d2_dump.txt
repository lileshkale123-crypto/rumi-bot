"""Flood extras: /flood /setfloodtimer /floodmode /clearflood + punishment used by flood watcher (aiogram 3.x)."""
import time
from typing import Any, Awaitable, Callable

import asyncpg
from aiogram import BaseMiddleware, Bot, Router
from aiogram.filters import Command
from aiogram.types import Message, TelegramObject

import ax_actions
import ax_mod
from ax_common import cmd_args, is_admin
from ax_warn2 import fmt_time, parse_time

router = Router(name="ax_flood")

MODES = ("ban", "mute", "kick", "tban", "tmute")
_recent: dict[tuple[int, int], list[tuple[float, int]]] = {}  # (chat_id, user_id) -> [(time, message_id)]
_RECENT_MAX = 10_000
_middleware_registered = False


def _prune_recent(now: float) -> None:
    if len(_recent) <= _RECENT_MAX:
        return
    stale = [k for k, lst in _recent.items() if not lst or now - lst[-1][0] >= 120.0]
    for k in stale:
        _recent.pop(k, None)
    if len(_recent) > _RECENT_MAX:
        oldest = sorted(_recent.items(), key=lambda kv: kv[1][-1][0] if kv[1] else 0.0)[: _RECENT_MAX // 5]
        for k, _ in oldest:
            _recent.pop(k, None)


def record_message(msg: Message) -> None:
    chat = msg.chat
    user = msg.from_user
    if chat.type == "private" or user is None or user.is_bot:
        return
    k = (chat.id, user.id)
    now = time.time()
    _prune_recent(now)
    lst = [x for x in _recent.get(k, []) if now - x[0] < 120.0][-49:]
    lst.append((now, msg.message_id))
    _recent[k] = lst


class FloodRecordMiddleware(BaseMiddleware):
    async def __call__(
        self,
        handler: Callable[[TelegramObject, dict[str, Any]], Awaitable[Any]],
        event: TelegramObject,
        data: dict[str, Any],
    ) -> Any:
        if isinstance(event, Message):
            record_message(event)
        return await handler(event, data)


async def get(pool: asyncpg.Pool, chat_id: int) -> dict:
    async with pool.acquire() as c:
        r = await c.fetchrow(
            "SELECT flood_limit, flood_window, flood_mute, flood_mode, flood_mode_secs, flood_clear "
            "FROM ax_settings WHERE chat_id=$1",
            chat_id,
        )
    if r:
        return dict(r)
    return {
        "flood_limit": 0,
        "flood_window": 5,
        "flood_mute": 300,
        "flood_mode": "tmute",
        "flood_mode_secs": 0,
        "flood_clear": False,
    }


async def punish(bot: Bot, pool: asyncpg.Pool, chat, user, msg: Message, mute_secs: int) -> None:
    s = await get(pool, chat.id)
    if s["flood_clear"]:
        raw_ids = [m for _, m in _recent.pop((chat.id, user.id), [])]
        if msg.message_id not in raw_ids:
            raw_ids.append(msg.message_id)
        # Deduplicate while preserving order and delete in Telegram's max 100-ID batches
        ids = list(dict.fromkeys(raw_ids))
        for i in range(0, len(ids), 100):
            try:
                await bot.delete_messages(chat.id, ids[i : i + 100])
            except Exception:
                pass
    mode, secs = s["flood_mode"], s["flood_mode_secs"]
    if mode in ("tban", "tmute"):
        action = f"{mode} {secs or (mute_secs if mode == 'tmute' else 3600)}s"
    else:
        action = mode
    await ax_actions.run(bot, pool, chat.id, user, action, "Flooding the chat")


async def _gate(message: Message, bot: Bot) -> bool:
    chat = message.chat
    user = message.from_user
    if user is None:
        return False
    if chat.type == "private":
        await ax_mod.say(
            message,
            "⚠️ <b>Groups only</b>\n<blockquote>Use this inside your group 💫</blockquote>",
            bot=bot,
        )
        return False
    if not await is_admin(bot, chat.id, user.id, "can_change_info"):
        await ax_mod.say(
            message,
            "🛡️ <b>Admins only</b>\n<blockquote>You need admin rights for this ✨</blockquote>",
            bot=bot,
        )
        return False
    return True


def _invalidate_settings_cache(chat_id: int) -> None:
    try:
        import ax_settings

        ax_settings._cache.pop(chat_id, None)
    except Exception:
        pass


@router.message(Command("flood"))
async def flood_cmd(message: Message, bot: Bot, pool: asyncpg.Pool) -> None:
    if not await _gate(message, bot):
        return
    s = await get(pool, message.chat.id)
    if s["flood_limit"] == 0:
        await ax_mod.say(
            message,
            "🌊 <b>Antiflood is off</b>\n<blockquote>Turn on: /setflood 6 (messages) or /setfloodtimer 10 30s</blockquote>",
            bot=bot,
        )
        return
    mode = s["flood_mode"]
    if mode in ("tban", "tmute"):
        mode += f" {fmt_time(s['flood_mode_secs'] or s['flood_mute'])}"
    await ax_mod.say(
        message,
        (
            "🌊 <b>Antiflood</b>\n<blockquote>"
            f"📊 <b>Trigger:</b> {s['flood_limit']} messages in {s['flood_window']}s\n"
            f"🔨 <b>Action:</b> {mode}\n"
            f"🧹 <b>Delete flood:</b> {'on' if s['flood_clear'] else 'off'}</blockquote>"
        ),
        bot=bot,
    )


@router.message(Command("setfloodtimer"))
async def setfloodtimer_cmd(message: Message, bot: Bot, pool: asyncpg.Pool) -> None:
    if not await _gate(message, bot):
        return
    chat_id = message.chat.id
    args = cmd_args(message)
    if args and args[0].lower() in ("off", "no"):
        import ax_settings

        await ax_settings.set_limit(pool, chat_id, 0)
        await ax_mod.say(
            message,
            "🌊 <b>Antiflood off</b>\n<blockquote>No flood checks now</blockquote>",
            bot=bot,
        )
        return
    secs = parse_time(args[1]) if len(args) > 1 else None
    if len(args) < 2 or not args[0].isdigit() or int(args[0]) < 3 or not secs or secs > 600:
        await ax_mod.say(
            message,
            "💫 <b>Usage</b>\n<blockquote>/setfloodtimer 10 30s: 10 messages within 30 seconds\n/setfloodtimer off</blockquote>",
            bot=bot,
        )
        return
    async with pool.acquire() as c:
        await c.execute(
            "INSERT INTO ax_settings (chat_id, flood_limit, flood_window) VALUES ($1,$2,$3) "
            "ON CONFLICT (chat_id) DO UPDATE SET flood_limit=$2, flood_window=$3",
            chat_id,
            int(args[0]),
            secs,
        )
    _invalidate_settings_cache(chat_id)
    await ax_mod.say(
        message,
        f"✅ <b>Timed antiflood set</b>\n<blockquote>{args[0]} messages within {fmt_time(secs)} triggers it 🌊</blockquote>",
        bot=bot,
    )


@router.message(Command("floodmode"))
async def floodmode_cmd(message: Message, bot: Bot, pool: asyncpg.Pool) -> None:
    if not await _gate(message, bot):
        return
    chat_id = message.chat.id
    args = [a.lower() for a in cmd_args(message)]
    if not args or args[0] not in MODES:
        s = await get(pool, chat_id)
        await ax_mod.say(
            message,
            f"⚙️ <b>Flood action</b>\n<blockquote>Currently <b>{s['flood_mode']}</b>\n"
            "Set: /floodmode ban, mute, kick, tban or tmute (example: /floodmode tmute 3h)</blockquote>",
            bot=bot,
        )
        return
    secs = 0
    if args[0] in ("tban", "tmute") and len(args) > 1:
        secs = parse_time(args[1])
        if not secs:
            await ax_mod.say(
                message,
                "⏳ <b>Bad time</b>\n<blockquote>Example: /floodmode tmute 3h</blockquote>",
                bot=bot,
            )
            return
    async with pool.acquire() as c:
        await c.execute(
            "INSERT INTO ax_settings (chat_id, flood_mode, flood_mode_secs) VALUES ($1,$2,$3) "
            "ON CONFLICT (chat_id) DO UPDATE SET flood_mode=$2, flood_mode_secs=$3",
            chat_id,
            args[0],
            secs,
        )
    _invalidate_settings_cache(chat_id)
    await ax_mod.say(
        message,
        f"✅ <b>Flood action set</b>\n<blockquote>Flooders get <b>{args[0]}</b></blockquote>",
        bot=bot,
    )

    await ax_mod.say(

    )

"""Flood protection and the settings menu (aiogram 3.x)."""
import time
from typing import Any, Awaitable, Callable

import asyncpg
from aiogram import BaseMiddleware, Bot, F, Router
from aiogram.enums import ParseMode
from aiogram.filters import Command
from aiogram.types import CallbackQuery, InlineKeyboardMarkup, Message, TelegramObject

import ax_welcome
from ax_common import btn, cmd_args, fire, is_admin

router = Router(name="ax_settings")
H = ParseMode.HTML

_cache: dict[int, tuple[float, tuple[int, int, int]]] = {}
_CACHE_MAX = 5000
_hits: dict[tuple[int, int], list[float]] = {}
_HITS_MAX = 10_000
_middleware_registered = False


def _prune_cache(now: float) -> None:
    if len(_cache) <= _CACHE_MAX:
        return
    stale = [cid for cid, (ts, _) in _cache.items() if now - ts >= 120.0]
    for cid in stale:
        _cache.pop(cid, None)
    if len(_cache) > _CACHE_MAX:
        oldest = sorted(_cache.items(), key=lambda kv: kv[1][0])[: _CACHE_MAX // 5]
        for cid, _ in oldest:
            _cache.pop(cid, None)


def _prune_hits(now: float) -> None:
    if len(_hits) <= _HITS_MAX:
        return
    stale = [k for k, lst in _hits.items() if not lst or now - lst[-1] >= 60.0]
    for k in stale:
        _hits.pop(k, None)
    if len(_hits) > _HITS_MAX:
        _hits.clear()


async def _cfg_load(pool: asyncpg.Pool, chat_id: int) -> tuple[int, int, int]:
    async with pool.acquire() as conn:
        r = await conn.fetchrow(
            "SELECT flood_limit, flood_window, flood_mute FROM ax_settings WHERE chat_id=$1",
            chat_id,
        )
    v = (int(r["flood_limit"]), int(r["flood_window"]), int(r["flood_mute"])) if r else (0, 5, 300)
    now = time.time()
    _prune_cache(now)
    _cache[chat_id] = (now, v)
    return v


async def cfg(pool: asyncpg.Pool, chat_id: int) -> tuple[int, int, int]:
    c = _cache.get(chat_id)
    if c:
        now = time.time()
        if now - c[0] >= 30.0:
            _cache[chat_id] = (now, c[1])  # only one refresh at a time
            fire(_cfg_load(pool, chat_id))
        return c[1]
    return await _cfg_load(pool, chat_id)


async def set_limit(pool: asyncpg.Pool, chat_id: int, n: int) -> None:
    async with pool.acquire() as conn:
        await conn.execute(
            "INSERT INTO ax_settings (chat_id, flood_limit) VALUES ($1,$2) "
            "ON CONFLICT (chat_id) DO UPDATE SET flood_limit=$2",
            chat_id,
            n,
        )
    _cache.pop(chat_id, None)


async def _check_flood(msg: Message, bot: Bot, pool: asyncpg.Pool) -> None:
    chat = msg.chat
    user = msg.from_user
    if chat.type == "private" or user is None or user.is_bot:
        return
    limit, window, mute = await cfg(pool, chat.id)
    if limit == 0:
        return
    now = time.time()
    _prune_hits(now)
    key = (chat.id, user.id)
    hits = [t for t in _hits.get(key, []) if now - t < window] + [now]
    _hits[key] = hits
    if len(hits) < limit:
        return
    _hits[key] = []

    if await is_admin(bot, chat.id, user.id):
        return
    import ax_approve

    if await ax_approve.is_approved(pool, chat.id, user.id):
        return
    try:
        import ax_flood

        await ax_flood.punish(bot, pool, chat, user, msg, mute)
    except Exception:
        pass


class FloodWatcherMiddleware(BaseMiddleware):
    """Non-blocking antiflood observer on group messages."""

    async def __call__(
        self,
        handler: Callable[[TelegramObject, dict[str, Any]], Awaitable[Any]],
        event: TelegramObject,
        data: dict[str, Any],
    ) -> Any:
        if isinstance(event, Message) and event.chat.type != "private":
            bot: Bot | None = data.get("bot")
            pool: asyncpg.Pool | None = data.get("pool")
            if bot is not None and pool is not None:
                await _check_flood(event, bot, pool)
        return await handler(event, data)

@router.callback_query(F.data.startswith("axs:"))
async def settings_callback(query: CallbackQuery, bot: Bot, pool: asyncpg.Pool) -> None:
    if query.message is None or not query.data:
        await query.answer()
        return
    chat = query.message.chat
    if not await is_admin(bot, chat.id, query.from_user.id, "can_change_info"):
        await query.answer("Admins only 🛡️", show_alert=True)
        return
    p = query.data.split(":")
    sec = p[1] if len(p) > 1 else "main"
    act = p[2] if len(p) > 2 else None

    if sec == "close":
        await query.answer()
        try:
            await query.message.delete()
        except Exception:
            pass
        return

    if sec == "flood":
        if act == "on" and len(p) > 3 and p[3].isdigit():
            await set_limit(pool, chat.id, int(p[3]))
        elif act == "off":
            await set_limit(pool, chat.id, 0)
        text, kb = await flood_view(pool, chat.id)
    elif sec in ("welcome", "goodbye"):
        if act in ("on", "off"):
            await ax_welcome.put(pool, chat.id, sec, "enabled", act == "on")
        elif act == "reset":
            await ax_welcome.put(pool, chat.id, sec, "text", None)
            await ax_welcome.put(pool, chat.id, sec, "photo", None)
        text, kb = await msg_view(pool, chat.id, sec)
    else:
        text, kb = main_view()

    await query.answer("Done ✨" if act else None)
    try:
        await query.message.edit_text(text, parse_mode=H, reply_markup=kb)
    except Exception:
        pass

"""Welcome and goodbye messages (aiogram 3.x)."""
import logging

import asyncpg
from aiogram import Bot, F, Router
from aiogram.enums import ParseMode
from aiogram.exceptions import TelegramBadRequest
from aiogram.filters import Command
from aiogram.types import LinkPreviewOptions, Message

import ax_fill

log = logging.getLogger(__name__)
router = Router(name="ax_welcome")

IMG = "assets/rumi-start-v2.png"
DEFAULTS = {
    "welcome": "✨ <b>Welcome, {first_name}!</b>\n<blockquote>Glad to have you in <b>{chat_name}</b> 💫\nBe kind and enjoy your stay 🌸</blockquote>",
    "goodbye": "👋 <b>Goodbye, {first_name}</b>\n<blockquote>See you again in <b>{chat_name}</b> 💫</blockquote>",
}
_VALID_KINDS = ("welcome", "goodbye")
_VALID_COLS = ("text", "enabled", "photo")


async def get(pool: asyncpg.Pool, chat_id: int, kind: str):
    assert kind in _VALID_KINDS
    async with pool.acquire() as c:
        r = await c.fetchrow(
            f"SELECT {kind}_text t, {kind}_enabled e, {kind}_photo p FROM ax_settings WHERE chat_id=$1",
            chat_id,
        )
    return (r["t"], r["e"], r["p"]) if r else (None, True, None)


async def get_full(pool: asyncpg.Pool, chat_id: int, kind: str):
    """Everything the welcome needs in a single query."""
    assert kind in _VALID_KINDS
    async with pool.acquire() as c:
        r = await c.fetchrow(
            f"SELECT {kind}_text t, {kind}_enabled e, {kind}_photo p, "
            f"clean_welcome cw, last_welcome_id lw FROM ax_settings WHERE chat_id=$1",
            chat_id,
        )
    if r:
        return r["t"], r["e"], r["p"], r["cw"], r["lw"]
    return None, True, None, None, None

def make_set(kind: str):
    async def cmd(message: Message, bot: Bot, pool: asyncpg.Pool) -> None:
        chat = message.chat
        user = message.from_user
        if user is None:
            return

        async def reply_html(t: str):
            return await message.reply(t, parse_mode=ParseMode.HTML)

        if chat.type == "private" or not await is_admin(bot, chat.id, user.id, "can_change_info"):
            await reply_html(
                "🛡️ <b>Admins only</b>\n<blockquote>Use this in your group as an admin ✨</blockquote>"
            )
            return

        raw_msg = message.text or message.caption or ""
        parts = raw_msg.split(maxsplit=1)
        arg = parts[1].strip() if len(parts) > 1 else ""
        rp = message.reply_to_message
        photo = message.photo[-1] if message.photo else (rp.photo[-1] if rp and rp.photo else None)
        if not arg and rp:
            arg = (rp.text or rp.caption or "").strip()
        low = arg.lower()

"""Captcha for new joiners: mute until solved, kick on timeout (aiogram 3.x + APScheduler)."""
import logging
import random
import time
from datetime import datetime, timedelta, timezone
from typing import Any, Awaitable, Callable

import asyncpg
from aiogram import BaseMiddleware, Bot, F, Router
from aiogram.enums import ParseMode
from aiogram.types import CallbackQuery, ChatPermissions, InlineKeyboardMarkup, Message, TelegramObject
from apscheduler.schedulers.asyncio import AsyncIOScheduler

import ax_approve
import ax_rules
from ax_common import btn, esc, fire, get_bot_username

log = logging.getLogger(__name__)
router = Router(name="ax_captcha")

SOLVED = ChatPermissions(
    can_send_messages=True,
    can_send_audios=True,
    can_send_documents=True,
    can_send_photos=True,
    can_send_videos=True,
    can_send_video_notes=True,
    can_send_voice_notes=True,
    can_send_polls=True,
    can_send_other_messages=True,
    can_add_web_page_previews=True,
    can_invite_users=True,
)
MUTED = ChatPermissions(can_send_messages=False)

_SC: dict[int, tuple[float, dict]] = {}
_SC_MAX = 5000
_SC_TTL = 20.0

# Fast in-memory index of (chat_id, user_id) awaiting text/math captcha answers
_pending_answers: set[tuple[int, int]] = set()
_pending_loaded = False
_middleware_registered = False

NEED = [True]


def _prune_sc(now: float) -> None:
    if len(_SC) <= _SC_MAX:
        return
    expired = [cid for cid, (ts, _) in _SC.items() if now - ts >= _SC_TTL * 5]
    for cid in expired:
        _SC.pop(cid, None)
    if len(_SC) > _SC_MAX:
        oldest = sorted(_SC.items(), key=lambda kv: kv[1][0])[: _SC_MAX // 5]
        for cid, _ in oldest:
            _SC.pop(cid, None)


async def _reload(pool: asyncpg.Pool, chat_id: int) -> dict:
    async with pool.acquire() as c:
        r = await c.fetchrow("SELECT * FROM ax_captcha_settings WHERE chat_id=$1", chat_id)
    v = (
        dict(r)
        if r
        else {
            "enabled": False,
            "mode": "button",
            "mute_secs": 0,
            "kick": True,
            "kick_secs": 300,
            "rules": False,
            "btn_text": None,
        }
    )
    now = time.time()
    _prune_sc(now)
    _SC[chat_id] = (now, v)
    return dict(v)


async def settings(pool: asyncpg.Pool, chat_id: int) -> dict:
    hit = _SC.get(chat_id)
    if hit:
        now = time.time()
        if now - hit[0] >= _SC_TTL:
            _SC[chat_id] = (now, hit[1])  # one refresh at a time
            fire(_reload(pool, chat_id))
        return dict(hit[1])
    return await _reload(pool, chat_id)


def challenge(mode: str) -> tuple[str | None, str | None]:
    if mode == "math":
        a, b = random.randint(2, 9), random.randint(2, 9)
        return f"{a} + {b} = ?", str(a + b)
    if mode == "text":
        w = "".join(random.choice("ABCDEFGHJKMNPQRSTUVWXYZ") for _ in range(4))
        return f"Type this code: <code>{w}</code>", w
    return None, None


async def join(msg: Message, bot: Bot, pool: asyncpg.Pool) -> None:
    chat = msg.chat
    if chat.type == "private" or not msg.new_chat_members:
        return
    s = await settings(pool, chat.id)
    if not s["enabled"]:
        return

    bot_username = await get_bot_username(bot)

    for m in msg.new_chat_members:
        if m.is_bot or m.id == bot.id:
            continue
        if await ax_approve.is_approved(pool, chat.id, m.id):
            continue
        try:
            await bot.restrict_chat_member(chat.id, m.id, permissions=MUTED)
        except Exception as e:
            log.warning("CAPTCHA mute failed for user %s in chat %s: %r", m.id, chat.id, e)
            continue  # continue checking remaining new members instead of aborting the batch
        try:
            q, ans = challenge(s["mode"])
            name = esc(m.first_name or "friend")
            rows = []
            if q:
                body = (
                    f"🔐 <b>Verify, {name}</b>\n<blockquote>{q}\n"
                    f"Send the answer here ✨</blockquote>"
                )
            else:
                body = (
                    f"🔐 <b>Welcome, {name}</b>\n<blockquote>Tap the "
                    f"button to prove you are human ✨</blockquote>"
                )
                label = s["btn_text"] or "✅ I am human"
                rows.append([btn(label, f"axc:{m.id}")])
            if s.get("rules") and bot_username:
                r_cfg = await ax_rules.get(pool, chat.id)
                r_label = r_cfg.get("button_text") or "📜 Rules"
                rows.append([btn(r_label, url=f"https://t.me/{bot_username}?start=rules_{chat.id}")])

            kb = InlineKeyboardMarkup(inline_keyboard=rows) if rows else None
            sent = await bot.send_message(
                chat.id,
                body,
                parse_mode=ParseMode.HTML,
                reply_markup=kb,
            )
            exp = datetime.now(timezone.utc) + timedelta(seconds=s["kick_secs"])
            async with pool.acquire() as c:
                await c.execute(
                    "INSERT INTO ax_captcha_pending VALUES ($1,$2,$3,$4,$5) "
                    "ON CONFLICT (chat_id,user_id) DO UPDATE SET answer=$3, "
                    "msg_id=$4, expires=$5",
                    chat.id,
                    m.id,
                    ans,
                    sent.message_id,
                    exp,
                )
            if ans:
                _pending_answers.add((chat.id, m.id))
            NEED[0] = True
        except Exception:
            log.exception("CAPTCHA send failed")


async def finish(bot: Bot, pool: asyncpg.Pool, chat_id: int, user_id: int, s: dict) -> bool:
    _pending_answers.discard((chat_id, user_id))
    async with pool.acquire() as c:
        r = await c.fetchrow(
            "DELETE FROM ax_captcha_pending WHERE chat_id=$1 AND "
            "user_id=$2 RETURNING msg_id",
            chat_id,
            user_id,
        )
    if not r:
        return False
    try:
        if s["mute_secs"] > 0:
            until = datetime.now(timezone.utc) + timedelta(seconds=s["mute_secs"])
            await bot.restrict_chat_member(chat_id, user_id, permissions=MUTED, until_date=until)
        else:
            await bot.restrict_chat_member(chat_id, user_id, permissions=SOLVED)
        if r["msg_id"]:
            await bot.delete_message(chat_id, r["msg_id"])
    except Exception:
        pass
    return True


@router.callback_query(F.data.regexp(r"^axc:\d+$"))
async def press(query: CallbackQuery, bot: Bot, pool: asyncpg.Pool) -> None:
    if query.message is None or not query.data:
        await query.answer()
        return
    uid = int(query.data.split(":")[1])
    if query.from_user.id != uid:
        await query.answer("This button is not for you 🌸", show_alert=True)
        return
    s = await settings(pool, query.message.chat.id)
    ok = await finish(bot, pool, query.message.chat.id, uid, s)
    await query.answer("Verified, welcome ✨" if ok else "Already done")


async def _handle_answer(msg: Message, bot: Bot, pool: asyncpg.Pool) -> bool:
    """Returns True if message was a captcha attempt and consumed."""
    chat = msg.chat
    user = msg.from_user
    if chat.type == "private" or user is None or not msg.text or msg.text.startswith("/"):
        return False
    key = (chat.id, user.id)
    if _pending_loaded and key not in _pending_answers:
        return False

    async with pool.acquire() as c:
        r = await c.fetchrow(
            "SELECT answer FROM ax_captcha_pending WHERE chat_id=$1 AND user_id=$2",
            chat.id,
            user.id,
        )
    if not r or not r["answer"]:
        _pending_answers.discard(key)
        return False

    try:
        await msg.delete()
    except Exception:
        pass

    if msg.text.strip().upper() == r["answer"].upper():
        s = await settings(pool, chat.id)
        await finish(bot, pool, chat.id, user.id, s)
    return True


class CaptchaMiddleware(BaseMiddleware):
    """Handles new-member captcha setup (while letting welcome run) and consumes text/math answers."""

    async def __call__(
        self,
        handler: Callable[[TelegramObject, dict[str, Any]], Awaitable[Any]],
        event: TelegramObject,
        data: dict[str, Any],
    ) -> Any:
        if isinstance(event, Message) and event.chat.type != "private":
            bot: Bot | None = data.get("bot")
            pool: asyncpg.Pool | None = data.get("pool")
            if bot is not None and pool is not None:
                if event.new_chat_members:
                    await join(event, bot, pool)
                elif event.text and not event.text.startswith("/"):
                    if await _handle_answer(event, bot, pool):
                        return None
        return await handler(event, data)


async def expire_job(bot: Bot, pool: asyncpg.Pool) -> None:
    if not NEED[0]:
        return
    try:
        async with pool.acquire() as c:
            rows = await c.fetch(
                "DELETE FROM ax_captcha_pending WHERE expires < NOW() "
                "RETURNING chat_id, user_id, msg_id"
            )
        for r in rows:
            _pending_answers.discard((r["chat_id"], r["user_id"]))
            s = await settings(pool, r["chat_id"])
            try:
                if r["msg_id"]:
                    await bot.delete_message(r["chat_id"], r["msg_id"])
                if s["kick"]:
                    await bot.ban_chat_member(r["chat_id"], r["user_id"])
                    await bot.unban_chat_member(r["chat_id"], r["user_id"])
            except Exception:
                pass
        async with pool.acquire() as c:
            if not await c.fetchval("SELECT count(*) FROM ax_captcha_pending"):
                NEED[0] = False
                _pending_answers.clear()
    except Exception:
        log.exception("captcha expire_job failed")


async def on_startup(bot: Bot, dispatcher) -> None:
    global _pending_loaded
    pool: asyncpg.Pool = dispatcher["pool"]
    scheduler: AsyncIOScheduler = dispatcher["scheduler"]

    try:
        async with pool.acquire() as c:
            rows = await c.fetch(
                "SELECT chat_id, user_id FROM ax_captcha_pending WHERE answer IS NOT NULL"
            )
        _pending_answers.clear()
        _pending_answers.update((r["chat_id"], r["user_id"]) for r in rows)
        _pending_loaded = True
    except Exception:
        log.exception("Failed loading pending captcha answers")

    scheduler.add_job(
        expire_job,
        trigger="interval",
        seconds=60,
        next_run_time=datetime.now(timezone.utc) + timedelta(seconds=20),
        args=[bot, pool],
        id="ax_captcha:expire_job",
        replace_existing=True,
    )


def register(dp) -> None:
    global _middleware_registered
    if not _middleware_registered:
        _middleware_registered = True
        dp.message.outer_middleware(CaptchaMiddleware())
    if router.parent_router is None:
        dp.include_router(router)

import ax_captcha
from ax_captcha import settings
from ax_common import cmd_args, esc, is_admin
from ax_warn2 import fmt_time, parse_time

router = Router(name="ax_captcha2")
H = ParseMode.HTML
_COLS = ("enabled", "mode", "mute_secs", "kick", "kick_secs", "rules", "btn_text")


async def _ok(message: Message, bot: Bot) -> bool:
    chat = message.chat
    user = message.from_user
    if user is None:
        return False
    if chat.type == "private" or not await is_admin(bot, chat.id, user.id, "can_change_info"):
        await message.reply(
            "🛡 <b>Admins only</b>\n<blockquote>Use this in your group as an admin ✨</blockquote>",
            parse_mode=H,
        )
        return False
    return True


async def _set(pool: asyncpg.Pool, chat_id: int, col: str, val) -> None:
    assert col in _COLS
    ax_captcha._SC.pop(chat_id, None)
    async with pool.acquire() as c:
        await c.execute(
            f"INSERT INTO ax_captcha_settings (chat_id, {col}) VALUES "
            f"($1,$2) ON CONFLICT (chat_id) DO UPDATE SET {col}=$2",
            chat_id,
            val,
        )


def _yn(a: str) -> bool | None:
    a = (a or "").lower()
    if a in ("on", "yes", "y", "true"):
        return True
    if a in ("off", "no", "n", "false"):
        return False
    return None


@router.message(Command("captcha"))
async def captcha(message: Message, bot: Bot, pool: asyncpg.Pool) -> None:
    if not await _ok(message, bot):
        return
    chat = message.chat
    args = cmd_args(message)
    v = _yn(args[0] if args else "")
    s = await settings(pool, chat.id)
    if v is None:
        await message.reply(
            f"🔐 <b>Captcha</b>\n<blockquote>Status: <b>"
            f"{'on' if s['enabled'] else 'off'}</b>\nMode: "
            f"<b>{s['mode']}</b>\nKick: <b>{'on' if s['kick'] else 'off'}"
            f"</b> after <b>{fmt_time(s['kick_secs'])}</b>\nMute after "
            f"solving: <b>{fmt_time(s['mute_secs']) if s['mute_secs'] else 'no'}"
            f"</b>\nUse /captcha on or off</blockquote>",
            parse_mode=H,
        )
        return
    await _set(pool, chat.id, "enabled", v)
    await message.reply(
        f"{'✅' if v else '🚫'} <b>Captcha {'on' if v else 'off'}</b>",
        parse_mode=H,
    )


@router.message(Command("captchamode"))
async def captchamode(message: Message, bot: Bot, pool: asyncpg.Pool) -> None:
    if not await _ok(message, bot):

    await message.reply(f"✅ <b>Mode: {a}</b>", parse_mode=H)

def _time(col: str, title: str, allow_zero: bool):
    async def cmd(message: Message, bot: Bot, pool: asyncpg.Pool) -> None:
        if not await _ok(message, bot):
            return
        args = cmd_args(message)
        a = args[0] if args else ""
        secs = 0 if (allow_zero and a == "0") else parse_time(a)
        if secs is None or (secs == 0 and not allow_zero):
            await message.reply(
                f"<blockquote>Use /{title} 5m (s, m, h, d, w)"
                f"{' or 0 for forever' if allow_zero else ''}</blockquote>",
                parse_mode=H,
            )
            return
        await _set(pool, message.chat.id, col, int(secs))
        await message.reply(
            f"✅ <b>{title}: {fmt_time(secs) if secs else 'off'}</b>",
            parse_mode=H,
        )

    return cmd

from typing import Any, Awaitable, Callable

import asyncpg
from aiogram import BaseMiddleware, Bot, Router
from aiogram.enums import ParseMode
from aiogram.filters import Command

from ax_common import cmd_args, get_bot_username, is_admin

router = Router(name="ax_cleancmd")
H = ParseMode.HTML
TYPES = ("admin", "user", "other")
_cache: dict[int, set[str]] = {}
_CACHE_MAX = 5000
_middleware_registered = False


async def kinds(pool: asyncpg.Pool, chat_id: int) -> set[str]:
    async with pool.acquire() as c:
        rows = await c.fetch("SELECT kind FROM ax_cleancmd WHERE chat_id=$1", chat_id)
    k = {r["kind"] for r in rows}
    if len(_cache) >= _CACHE_MAX and chat_id not in _cache:
        _cache.pop(next(iter(_cache)), None)
    _cache[chat_id] = k
    return k


async def _gate(message: Message, bot: Bot, pool: asyncpg.Pool):
    chat = message.chat
    user = message.from_user
    if user is None:
        return None
    if chat.type == "private" or not await is_admin(bot, chat.id, user.id, "can_change_info"):
        await message.reply(

def _parse(args: list[str]) -> set[str]:
    out: set[str] = set()
    for a in args:

        if a == "all":
            out |= set(TYPES)
        elif a in TYPES:

class CleanCmdMiddleware(BaseMiddleware):
    """Schedules deletion of group commands matching ax_cleancmd without blocking command execution."""

    async def __call__(
        self,
        handler: Callable[[TelegramObject, dict[str, Any]], Awaitable[Any]],
        event: TelegramObject,
        data: dict[str, Any],
    ) -> Any:
        if (
            isinstance(event, Message)
            and event.chat.type in ("group", "supergroup")
            and event.from_user is not None
            and event.text
            and event.text.startswith("/")
        ):
            bot: Bot | None = data.get("bot")
            pool: asyncpg.Pool | None = data.get("pool")
            scheduler: AsyncIOScheduler | None = data.get("scheduler")
            if bot is not None and pool is not None and scheduler is not None:
                k = _cache.get(event.chat.id)
                if k is None:
                    k = await kinds(pool, event.chat.id)
                if k:
                    first = event.text.split()[0]
                    uname = (data.get("bot_username") or await get_bot_username(bot)).lower()
                    me = "@" + uname if uname else ""
                    if "@" in first and (not me or not first.lower().endswith(me)):
                        kind = "other"
                    elif await is_admin(bot, event.chat.id, event.from_user.id):
                        kind = "admin"
                    else:
                        kind = "user"
                    if kind in k:
                        run_at = datetime.now(timezone.utc) + timedelta(seconds=3)
                        scheduler.add_job(
                            _del,
                            trigger="date",
                            run_date=run_at,
                            args=[bot, event.chat.id, event.message_id],
                            id=f"cleancmd:{event.chat.id}:{event.message_id}",
                            replace_existing=True,
                        )
        return await handler(event, data)


def register(dp) -> None:
    global _middleware_registered
    if not _middleware_registered:
        _middleware_registered = True
        dp.message.outer_middleware(CleanCmdMiddleware())
    if router.parent_router is None:
        dp.include_router(router)

def kind_of(m: Message) -> str | None:
    if m.new_chat_members:
        return "join"
    if m.left_chat_member:
        return "leave"
    if m.new_chat_photo or m.delete_chat_photo:
        return "photo"
    if m.pinned_message:
        return "pin"
    if m.new_chat_title:
        return "title"
    if (
        m.video_chat_started
        or m.video_chat_ended
        or m.video_chat_scheduled
        or m.video_chat_participants_invited
    ):
        return "videochat"
    if (
        m.group_chat_created
        or m.supergroup_chat_created
        or m.channel_chat_created
        or m.message_auto_delete_timer_changed
        or m.migrate_to_chat_id
        or m.migrate_from_chat_id
    ):
        return "other"
    return None

from typing import Any, Awaitable, Callable

import asyncpg
from aiogram import BaseMiddleware, Bot, Router
from aiogram.enums import ParseMode
from aiogram.filters import Command
from aiogram.types import Message, TelegramObject

from ax_common import cmd_args, esc, is_admin

PROTECTED = {
    "disable", "enable", "disabled", "disableable",
    "disabledel", "disableadmin", "start", "help",
}
_cache: dict[int, tuple[set[str], bool, bool]] = {}
_CACHE_MAX = 5000
_middleware_registered = False


async def load(pool: asyncpg.Pool, chat_id: int) -> tuple[set[str], bool, bool]:
    async with pool.acquire() as c:
        rows = await c.fetch("SELECT cmd FROM ax_disabled WHERE chat_id=$1", chat_id)
        s = await c.fetchrow(
            "SELECT del_msg, admin_too FROM ax_disable_settings WHERE chat_id=$1",
            chat_id,
        )
    d = (
        {r["cmd"] for r in rows},
        bool(s and s["del_msg"]),
        bool(s and s["admin_too"]),
    )
    if len(_cache) >= _CACHE_MAX and chat_id not in _cache:
        _cache.pop(next(iter(_cache)), None)
    _cache[chat_id] = d
    return d

    chat = message.chat
    user = message.from_user
    if user is None:
        return None
    if chat.type == "private" or not await is_admin(bot, chat.id, user.id, "can_change_info"):
        await message.reply(

            parse_mode=H,

def _flag(col: str, name: str):
    assert col in ("del_msg", "admin_too")

    async def cmd(message: Message, bot: Bot, pool: asyncpg.Pool) -> None:
        g = await _gate(message, bot, pool)
        if not g:
            return
        pool, cid = g
        args = cmd_args(message)
        a = args[0].lower() if args else ""
        if a not in ("on", "off", "yes", "no"):
            await message.reply(f"<blockquote>Use /{name} on or off</blockquote>", parse_mode=H)
            return
        v = a in ("on", "yes")
        async with pool.acquire() as c:
            await c.execute(
                f"INSERT INTO ax_disable_settings (chat_id, {col}) "
                f"VALUES ($1,$2) ON CONFLICT (chat_id) DO UPDATE SET {col}=$2",
                cid,
                v,
            )
        await load(pool, cid)
        await message.reply(
            f"{'✅' if v else '🚫'} <b>{name} {'on' if v else 'off'}</b>",
            parse_mode=H,
        )

    return cmd


class DisableMiddleware(BaseMiddleware):
    """Blocks disabled commands in groups before command routers run."""

    async def __call__(
        self,
        handler: Callable[[TelegramObject, dict[str, Any]], Awaitable[Any]],
        event: TelegramObject,
        data: dict[str, Any],
    ) -> Any:
        if (
            isinstance(event, Message)
            and event.chat.type in ("group", "supergroup")
            and event.from_user is not None

def _prune_cache(store: dict, now: float) -> None:
    if len(store) <= _CACHE_MAX:
        return
    expired = [cid for cid, (ts, _) in store.items() if now - ts >= _CACHE_TTL]
    for cid in expired:
        store.pop(cid, None)
    if len(store) > _CACHE_MAX:
        oldest = sorted(store.items(), key=lambda kv: kv[1][0])[: _CACHE_MAX // 5]
        for cid, _ in oldest:
            store.pop(cid, None)

    _filter_cache.pop(chat_id, None)
    _note_names_cache.pop(chat_id, None)


async def _get_filters(pool: asyncpg.Pool, chat_id: int) -> list[dict]:
    now = time.time()
    hit = _filter_cache.get(chat_id)
    if hit and now - hit[0] < _CACHE_TTL:
        return hit[1]

            "SELECT trigger_text, content_type, text_content, file_id FROM ax_filters WHERE chat_id=$1",
            chat_id,
        )

def extract(reply: Message | None, rest: str) -> tuple[str, str | None, str | None]:
    if reply:
        for k in KINDS:
            obj = getattr(reply, k, None)
            if obj:
                fid = obj[-1].file_id if k == "photo" else obj.file_id
                text = None if k == "sticker" else (reply.caption or rest or None)
                return k, text, fid
        if reply.text and not rest:
            return "text", reply.text, None
    return "text", rest or None, None


async def deliver(

    text: str | None,
    fid: str | None,

def match_trigger(trig: str, low: str) -> bool:
    if trig.startswith("exact:"):
        return low.strip() == trig[6:]
    if trig.startswith("prefix:"):
        return low.startswith(trig[7:])
    return re.search(r"(?<!\w)" + re.escape(trig) + r"(?!\w)", low) is not None

class NotesFiltersMiddleware(BaseMiddleware):
    """Non-blocking watcher for group filters and #notename triggers."""

    async def __call__(
        self,
        handler: Callable[[TelegramObject, dict[str, Any]], Awaitable[Any]],
        event: TelegramObject,
        data: dict[str, Any],
    ) -> Any:
        if (
            isinstance(event, Message)
            and event.chat.type in ("group", "supergroup")
            and event.text
            and event.from_user is not None
            and not event.from_user.is_bot
        ):
            bot: Bot | None = data.get("bot")
            pool: asyncpg.Pool | None = data.get("pool")
            if bot is not None and pool is not None:
                if not event.text.startswith("/"):
                    await _run_filter_watcher(event, bot, pool)
                if event.text.startswith("#"):
                    await _run_hash_watcher(event, bot, pool)
        return await handler(event, data)

class LinkedChannelMiddleware(BaseMiddleware):
    """Handles automatic forwards from linked channels (unpin and/or delete)."""

    async def __call__(
        self,
        handler: Callable[[TelegramObject, dict[str, Any]], Awaitable[Any]],
        event: TelegramObject,
        data: dict[str, Any],
    ) -> Any:
        if isinstance(event, Message) and event.is_automatic_forward:
            bot: Bot | None = data.get("bot")
            pool: asyncpg.Pool | None = data.get("pool")
            if bot is not None and pool is not None:
                s = await _settings(pool, event.chat.id)
                deleted = False
                try:
                    if s["antichannelpin"]:
                        await bot.unpin_chat_message(
                            chat_id=event.chat.id,
                            message_id=event.message_id,
                        )
                    if s["cleanlinked"]:
                        await event.delete()
                        deleted = True
                except Exception as e:
                    log.warning("linked watch failed: %s", e)
                if deleted:
                    return None
        return await handler(event, data)


def register(dp) -> None:
    global _middleware_registered
    if not _middleware_registered:
        _middleware_registered = True
        dp.message.outer_middleware(LinkedChannelMiddleware())
    if router.parent_router is None:
        dp.include_router(router)

_CACHE_TTL = 60.0
_CACHE_MAX = 5000


def _prune_cache(now: float) -> None:
    if len(_cache) <= _CACHE_MAX:
        return
    expired = [cid for cid, (ts, _, _) in _cache.items() if now - ts >= _CACHE_TTL]
    for cid in expired:
        _cache.pop(cid, None)

async def _row(pool: asyncpg.Pool, chat_id: int) -> tuple[int | None, set[str]]:
    now = time.time()
    hit = _cache.get(chat_id)
    if hit and now - hit[0] < _CACHE_TTL:
        return hit[1], set(hit[2])
    async with pool.acquire() as c:
        r = await c.fetchrow(
            "SELECT log_chat, off_cats FROM ax_log_settings WHERE chat_id=$1",
            chat_id,
        )
    if not r:
        lc, off = None, set()
    else:
        lc = r["log_chat"]
        off = {x for x in (r["off_cats"] or "").split(",") if x}
    _prune_cache(now)
    _cache[chat_id] = (now, lc, set(off))
    return lc, off


async def send_log(bot: Bot, pool: asyncpg.Pool, chat_id: int, cat: str, text: str) -> None:
    """Send one log line. Never raises."""
    try:
        log_chat, off = await _row(pool, chat_id)
        if not log_chat or cat in off:
            return
        title = ""
        try:
            title = (await bot.get_chat(chat_id)).title or ""
        except Exception:
            pass
        await bot.send_message(
            log_chat,
            f"📋 <b>{esc(title)}</b> · <i>{cat}</i>\n<blockquote>{text}</blockquote>",
            parse_mode=H,
        )
    except Exception:
        pass

"""AFK status (long notes, stickers, GIFs, voice notes, photos, videos) and /info (aiogram 3.x)."""
import asyncio
import logging
import time
from datetime import datetime, timezone
from typing import Any, Awaitable, Callable

import asyncpg
from aiogram import BaseMiddleware, Bot, Router
from aiogram.enums import ParseMode
from aiogram.filters import Command
from aiogram.types import Message, TelegramObject

import config
import ui
from ax_common import cmd_args, esc, find_user_by_username, resolve_target

log = logging.getLogger(__name__)
router = Router(name="ax_afk")

H = ParseMode.HTML
_notice: dict[tuple[int, int], float] = {}
_NOTICE_MAX = 5000

MAXLEN = 1024        # longest AFK note (characters)
CAPTION = 1024       # Telegram's caption limit for GIF / voice / photo / video
TEXT_LIMIT = 4000    # Telegram's message limit is 4096
LABELS = {
    "sticker": "Sticker",
    "animation": "GIF",
    "voice": "Voice note",
    "photo": "Photo",
    "video": "Video",
}

ROLES = {
    "creator": "👑 Owner",
    "administrator": "🛡️ Admin",
    "member": "👤 Member",
    "restricted": "🚫 Restricted",
    "left": "🚪 Left",
    "kicked": "⛔ Banned",
}

_afk_ids: set[int] = set()
_afk_loaded = False
_load_lock = asyncio.Lock()
_middleware_registered = False


def dur(start: datetime) -> str:
    if start.tzinfo is None:
        start = start.replace(tzinfo=timezone.utc)
    s = max(0, int((datetime.now(timezone.utc) - start).total_seconds()))
    h, r = divmod(s, 3600)
    m, sec = divmod(r, 60)
    parts = ([f"{h}h"] if h else []) + ([f"{m}m"] if m else []) + [f"{sec}s"]
    return " ".join(parts)


def media_of(m: Message | None) -> tuple[str | None, str | None]:
    """(type, file_id) of the first supported media in a message, else (None, None)."""
    if m is None:
        return None, None
    if m.sticker:
        return "sticker", m.sticker.file_id
    if m.animation:  # GIFs (check before video)
        return "animation", m.animation.file_id
    if m.voice:
        return "voice", m.voice.file_id
    if m.photo:
        return "photo", m.photo[-1].file_id
    if m.video:
        return "video", m.video.file_id
    return None, None


def quote(inner: str, reason: str | None) -> str:
    """Long notes go in a collapsed (tap to expand) quote so the chat stays tidy."""
    tag = "blockquote expandable" if reason and len(reason) > 300 else "blockquote"
    return f"<{tag}>{inner}</blockquote>"


def fit(build, reason: str | None, limit: int) -> str:
    """build(reason) -> html. Shortens the reason until the whole text fits `limit`."""
    r = reason
    text = build(r)
    while len(text) > limit and r:
        r = r[: max(0, int(len(r) * 0.8) - 1)].rstrip()
        text = build((r + "…") if r else None)
    return text


def afk_text(name: str, start: datetime, reason: str | None) -> str:
    extra = f"\n📝 {esc(reason)}" if reason else ""
    return (
        f"💤 <b>{esc(name)} is AFK</b>\n"
        + quote(f"⏳ Away for <b>{dur(start)}</b>{extra}", reason)
    )


_SEND = {
    "animation": lambda m, f, c: m.reply_animation(animation=f, caption=c, parse_mode=H),
    "voice": lambda m, f, c: m.reply_voice(voice=f, caption=c, parse_mode=H),
    "photo": lambda m, f, c: m.reply_photo(photo=f, caption=c, parse_mode=H),
    "video": lambda m, f, c: m.reply_video(video=f, caption=c, parse_mode=H),
}


async def send_notice(msg: Message, mtype: str | None, mid: str | None, name: str, start: datetime, reason: str | None) -> None:
    """Tell the group that someone is AFK, with their note and attached sticker/GIF/voice."""
    build = lambda r: afk_text(name, start, r)
    if mtype in _SEND and mid:
        try:  # media with the note as its caption
            await _SEND[mtype](msg, mid, fit(build, reason, CAPTION))
            return
        except Exception:
            pass  # e.g. group blocks media: fall back to text
    elif mtype == "sticker" and mid:
        try:  # stickers can't have captions: sticker, then text
            await msg.reply_sticker(sticker=mid)
        except Exception:
            pass
    await msg.reply(fit(build, reason, TEXT_LIMIT), parse_mode=H)


def _prune_notice(now: float) -> None:
    if len(_notice) <= _NOTICE_MAX:
        return
    stale = [k for k, ts_val in _notice.items() if now - ts_val >= 60.0]
    for k in stale:
        _notice.pop(k, None)
    if len(_notice) > _NOTICE_MAX:
        _notice.clear()


async def _load_afk(pool: asyncpg.Pool) -> None:
    global _afk_loaded
    if _afk_loaded:
        return
    async with _load_lock:
        if _afk_loaded:
            return
        async with pool.acquire() as c:
            await c.execute("ALTER TABLE ax_afk ADD COLUMN IF NOT EXISTS media_type TEXT")
            await c.execute("ALTER TABLE ax_afk ADD COLUMN IF NOT EXISTS media_id TEXT")
            rows = await c.fetch("SELECT user_id FROM ax_afk")
        _afk_ids.clear()
        _afk_ids.update(r["user_id"] for r in rows)
        _afk_loaded = True


@router.message(Command("afk"))
async def afk_cmd(message: Message, pool: asyncpg.Pool) -> None:
    user = message.from_user
    if user is None:
        return
    try:
        await _load_afk(pool)
    except Exception as e:
        log.warning("afk load error: %s", e)

    parts = (message.text or "").split(None, 1)  # keep the line breaks of a long note
    arg = parts[1].strip() if len(parts) > 1 else ""
    if arg.lower() == "off":
        _afk_ids.discard(user.id)
        async with pool.acquire() as c:
            r = await c.execute("DELETE FROM ax_afk WHERE user_id=$1", user.id)
        await message.reply(
            "✅ <b>AFK removed</b>" if r.endswith("1") else "💫 <b>You're not AFK</b>",
            parse_mode=H,
        )
        return

    mtype, mid = media_of(message.reply_to_message)  # reply to a sticker/GIF/voice to attach it
    reason = arg[:MAXLEN] or None
    _afk_ids.add(user.id)
    async with pool.acquire() as c:
        await c.execute(
            "INSERT INTO ax_afk (user_id, reason, media_type, media_id) VALUES ($1,$2,$3,$4) "
            "ON CONFLICT (user_id) DO UPDATE SET reason=$2, media_type=$3, media_id=$4, "
            "start_time=now(), msg_count=0",
            user.id,
            reason,
            mtype,
            mid,
        )

    def build(r):
        extra = (f"\n📝 {esc(r)}" if r else "") + (f"\n📎 {LABELS[mtype]} attached" if mtype else "")
        return (
            "💤 <b>AFK on</b>\n"
            + quote(f"👤 <b>{esc(user.first_name or 'User')}</b> is away{extra}", r)
        )

    await message.reply(fit(build, reason, TEXT_LIMIT), parse_mode=H)


async def _handle_afk_watcher(msg: Message, pool: asyncpg.Pool) -> None:
    user = msg.from_user
    chat = msg.chat
    if user is None or user.is_bot or chat.type == "private":
        return

    raw = msg.text or msg.caption or ""
    try:
        await _load_afk(pool)
    except Exception as e:
        log.warning("afk load error: %s", e)
        return
    if not _afk_ids:
        return

    if not raw.lower().startswith("/afk") and user.id in _afk_ids:
        _afk_ids.discard(user.id)
        async with pool.acquire() as c:
            r = await c.fetchrow(
                "DELETE FROM ax_afk WHERE user_id=$1 RETURNING reason, start_time, msg_count",
                user.id,
            )
        if r:
            note = r["reason"] or ""
            short = note[:200] + ("…" if len(note) > 200 else "")
            extra = f"\n📝 {esc(short)}" if short else ""
            try:
                await msg.reply(
                    f"👋 <b>Welcome back, {esc(user.first_name or 'friend')}!</b>\n<blockquote>"
                    f"⏳ Away for <b>{dur(r['start_time'])}</b>{extra}\n"
                    f"💌 Mentions while away: <b>{r['msg_count']}</b></blockquote>",
                    parse_mode=H,
                )
            except Exception:
                pass

    if raw.startswith("/"):
        return

    targets: set[int] = set()
    rp = msg.reply_to_message
    if rp and rp.from_user and not rp.from_user.is_bot:
        targets.add(rp.from_user.id)

    su = getattr(getattr(msg, "forward_origin", None), "sender_user", None)
    if su and not su.is_bot:
        targets.add(su.id)

    ents = msg.entities if msg.text else (msg.caption_entities or [])
    src_text = msg.text if msg.text else (msg.caption or "")
    for e in ents or []:
        if e.type == "text_mention" and e.user and not e.user.is_bot:
            targets.add(e.user.id)
        elif e.type == "mention" and src_text:
            name = e.extract_from(src_text).lstrip("@")
            if name:
                row = await find_user_by_username(pool, name)
                if row:
                    targets.add(row["id"])

    now = time.time()
    _prune_notice(now)
    for tid in (targets & _afk_ids) - {user.id}:
        async with pool.acquire() as c:  # one round trip: count the mention and fetch everything
            r = await c.fetchrow(
                "WITH u AS (UPDATE ax_afk SET msg_count = msg_count + 1 WHERE user_id=$1 "
                "RETURNING reason, start_time, msg_count, media_type, media_id) "
                "SELECT u.reason, u.start_time, u.msg_count, u.media_type, u.media_id, "
                "(SELECT first_name FROM users WHERE id=$1) AS first_name FROM u",
                tid,
            )
        if not r:
            _afk_ids.discard(tid)
            continue
        key = (chat.id, tid)
        if now - _notice.get(key, 0.0) < 15.0:
            continue
        _notice[key] = now
        try:
            await send_notice(
                msg,
                r["media_type"],
                r["media_id"],
                r["first_name"] or "This user",
                r["start_time"],
                r["reason"],
            )
        except Exception:
            pass


class AfkMiddleware(BaseMiddleware):
    """Non-blocking watcher for AFK returns and AFK mentions in groups."""

    async def __call__(
        self,
        handler: Callable[[TelegramObject, dict[str, Any]], Awaitable[Any]],
        event: TelegramObject,
        data: dict[str, Any],
    ) -> Any:
        if isinstance(event, Message) and event.chat.type != "private":
            pool: asyncpg.Pool | None = data.get("pool")
            if pool is not None:
                await _handle_afk_watcher(event, pool)
        return await handler(event, data)


def ts(t: datetime | None) -> str:
    return t.strftime("%d %b %Y, %H:%M UTC") if t else "—"


@router.message(Command("info"))
async def info_cmd(message: Message, bot: Bot, pool: asyncpg.Pool) -> None:

def _invalidate_all_caches(chat_id: int) -> None:
    """Clear per-chat settings caches after import/reset so changes take effect immediately."""
    for mod_name, fn_or_dict in (
        ("ax_lockcore", "invalidate"),
        ("ax_blocklist", "invalidate"),
        ("ax_notes", "invalidate"),
        ("ax_rules", "invalidate"),
        ("ax_pins", "invalidate"),
        ("ax_reports", "invalidate"),
        ("ax_log", "invalidate"),
    ):
        try:
            mod = __import__(mod_name)
            fn = getattr(mod, fn_or_dict, None)
            if callable(fn):
                fn(chat_id)
        except Exception:
            pass
    for mod_name, attr in (
        ("ax_settings", "_cache"),
        ("ax_raid", "_SC"),
        ("ax_captcha", "_SC"),
        ("ax_cleancmd", "_cache"),
        ("ax_cleanservice", "_cache"),
        ("ax_disable", "_cache"),
        ("ax_warn", "_SCACHE"),
    ):
        try:
            mod = __import__(mod_name)
            store = getattr(mod, attr, None)
            if isinstance(store, dict):
                store.pop(chat_id, None)
        except Exception:
            pass

@router.message(Command("import"))
async def import_cmd(message: Message, bot: Bot, pool: asyncpg.Pool) -> None:
    chat = message.chat
    user = message.from_user
    if user is None:
        return
    if chat.type not in ("group", "supergroup"):
        await _say(message, "<b>Groups only</b>\n<blockquote>Use this inside a group ✨</blockquote>")
        return
    if not await _is_creator(bot, chat.id, user.id):
        await _say(message, "👑 <b>Creator only</b>\n<blockquote>Only the group creator can import ✨</blockquote>")
        return

    doc = message.reply_to_message.document if message.reply_to_message else None
    if not doc or (doc.file_size or 0) > 1_000_000:
        await _say(
            message,
            "📎 <b>Reply to a backup file</b>\n<blockquote>Reply to a Rumi backup .json file with /import 💫</blockquote>",
        )
        return
    try:
        f = await bot.get_file(doc.file_id)
        buf = io.BytesIO()
        await bot.download_file(f.file_path, destination=buf)
        d = json.loads(buf.getvalue().decode("utf-8"))
        tabs = d["tables"]
        assert d.get("rumi_backup") == 1 and isinstance(tabs, dict)
    except Exception:
        await _say(
            message,
            "⚠️ <b>Not a Rumi backup</b>\n<blockquote>That file could not be read ✨</blockquote>",
        )
        return

    done = 0
    try:
        async with pool.acquire() as c:
            async with c.transaction():
                for t in TABLES:
                    if t not in tabs:
                        continue
                    got = await c.fetch(
                        "SELECT column_name FROM information_schema.columns WHERE table_name=$1",
                        t,
                    )
                    cols = {r["column_name"] for r in got}
                    await c.execute("DELETE FROM %s WHERE chat_id=$1" % t, chat.id)
                    for row in tabs[t]:
                        keys = [k for k in row if k in cols and k not in SKIP]
                        names = ",".join(["chat_id"] + keys)
                        ph = ",".join("$%d" % (i + 1) for i in range(len(keys) + 1))
                        await c.execute(
                            "INSERT INTO %s (%s) VALUES (%s) ON CONFLICT DO NOTHING" % (t, names, ph),
                            chat.id,
                            *[row[k] for k in keys],
                        )
                        done += 1
    except Exception as e:
        await _say(
            message,
            "⚠️ <b>Import failed</b>\n<blockquote>Nothing was changed: %s</blockquote>" % esc(str(e)[:150]),
        )
        return

    _invalidate_all_caches(chat.id)
    await _say(
        message,
        "✅ <b>Imported</b>\n<blockquote>%d entries loaded, %s 💫</blockquote>"
        % (done, esc(user.first_name or "friend")),
    )

from aiogram import BaseMiddleware, Bot
from aiogram.enums import ParseMode
from aiogram.types import CallbackQuery, Chat, Message, TelegramObject
from pydantic import PrivateAttr

from ax_common import is_admin

log = logging.getLogger(__name__)
H = ParseMode.HTML

CMDS = {
    "save", "get", "notes", "clear", "clearallnotes",
    "filter", "filters", "stop", "clearfilters",
    "lock", "unlock", "locks", "rules", "setrules", "resetrules",
    "warnings", "warnlimit", "warnmode", "warntime",
    "welcome", "setwelcome", "resetwelcome",
    "goodbye", "setgoodbye", "resetgoodbye",
    "setflood", "flood", "settings",
}

_conn_cache: dict[int, tuple[float, int | None]] = {}
_CONN_TTL = 60.0
_CONN_MAX = 5000
_middleware_registered = False


def invalidate(user_id: int) -> None:
    _conn_cache.pop(user_id, None)


def _prune_conn_cache(now: float) -> None:
    if len(_conn_cache) <= _CONN_MAX:
        return
    expired = [uid for uid, (ts, _) in _conn_cache.items() if now - ts >= _CONN_TTL]
    for uid in expired:
        _conn_cache.pop(uid, None)
    if len(_conn_cache) > _CONN_MAX:
        oldest = sorted(_conn_cache.items(), key=lambda kv: kv[1][0])[: _CONN_MAX // 5]
        for uid, _ in oldest:
            _conn_cache.pop(uid, None)


async def _connected(pool: asyncpg.Pool, user_id: int) -> int | None:
    now = time.time()
    hit = _conn_cache.get(user_id)
    if hit and now - hit[0] < _CONN_TTL:
        return hit[1]
    cid = None
    try:
        async with pool.acquire() as c:
            r = await c.fetchrow("SELECT chat_id FROM ax_connections WHERE user_id=$1", user_id)
        if r and r["chat_id"]:
            cid = int(r["chat_id"])
    except Exception:
        pass
    _prune_conn_cache(now)
    _conn_cache[user_id] = (now, cid)
    return cid


def _extract_cmd(message: Message) -> str | None:
    raw = (message.text or message.caption or "").strip()
    if not raw.startswith("/"):
        return None
    first = raw.split(None, 1)[0][1:]
    return first.split("@", 1)[0].lower()


class _DMProxyMessage(Message):
    """Message subclass whose .chat points to the connected group while all replies/edits target the DM."""

    _orig_msg: Message = PrivateAttr()

    def answer(self, *args, **kwargs):

def _proxy_message(msg: Message, cid: int, title: str | None = None) -> _DMProxyMessage:
    fake_chat = Chat(id=cid, type="supergroup", title=title, is_forum=False, username=None)
    field_data = {name: getattr(msg, name) for name in Message.model_fields}
    field_data["chat"] = fake_chat
    proxy = _DMProxyMessage.model_construct(_fields_set=set(msg.model_fields_set), **field_data)
    proxy._orig_msg = msg
    if msg.bot is not None:
        proxy.as_(msg.bot)
    return proxy


class DMProxyMessageMiddleware(BaseMiddleware):
    """Proxies DM commands in CMDS so they operate on the user's connected group."""

    async def __call__(
        self,
        handler: Callable[[TelegramObject, dict[str, Any]], Awaitable[Any]],
        event: TelegramObject,
        data: dict[str, Any],
    ) -> Any:
        if (
            isinstance(event, Message)
            and event.chat.type == "private"
            and event.from_user is not None
        ):
            cmd = _extract_cmd(event)
            if cmd in CMDS:
                bot: Bot | None = data.get("bot")
                pool: asyncpg.Pool | None = data.get("pool")
                if bot is not None and pool is not None:
                    cid = await _connected(pool, event.from_user.id)
                    if not cid:
                        await event.reply(
                            "\U0001F517 <b>Connect a group first</b>\n"
                            "<blockquote>Send /connect inside your group and tap the button. "
                            "After that, these commands work here and change that group \u2728</blockquote>",
                            parse_mode=H,
                        )
                        return None
                    if not await is_admin(bot, cid, event.from_user.id):
                        await event.reply(
                            "\U0001F512 <b>Admins only</b>\n"
                            "<blockquote>You are no longer an admin of the connected group. "
                            "Use /disconnect or /connect again \u2728</blockquote>",
                            parse_mode=H,
                        )
                        return None
                    title = None
                    try:
                        title = await pool.fetchval("SELECT title FROM ax_groups WHERE chat_id=$1", cid)
                    except Exception:
                        pass
                    proxy = _proxy_message(event, cid, title=title)
                    return await handler(proxy, data)
        return await handler(event, data)


class DMProxyCallbackMiddleware(BaseMiddleware):
    """Proxies DM ^axs: settings callbacks so they operate on the user's connected group."""

    async def __call__(
        self,
        handler: Callable[[TelegramObject, dict[str, Any]], Awaitable[Any]],
        event: TelegramObject,
        data: dict[str, Any],
    ) -> Any:
        if (
            isinstance(event, CallbackQuery)
            and event.data
            and event.data.startswith("axs:")
            and isinstance(event.message, Message)
            and event.message.chat.type == "private"
        ):
            bot: Bot | None = data.get("bot")
            pool: asyncpg.Pool | None = data.get("pool")
            if bot is not None and pool is not None:
                cid = await _connected(pool, event.from_user.id)
                if not cid:
                    await event.answer("Send /connect in your group first", show_alert=True)
                    return None
                if not await is_admin(bot, cid, event.from_user.id):
                    await event.answer("You are no longer an admin there", show_alert=True)
                    return None
                proxy_msg = _proxy_message(event.message, cid)
                proxy_q = event.model_copy(update={"message": proxy_msg})
                if event.bot is not None:
                    proxy_q.as_(event.bot)
                return await handler(proxy_q, data)
        return await handler(event, data)


def register(dp) -> None:
    global _middleware_registered
    if _middleware_registered:
        return
    _middleware_registered = True
    dp.message.outer_middleware(DMProxyMessageMiddleware())
    dp.callback_query.outer_middleware(DMProxyCallbackMiddleware())
    log.info("DM proxy middleware registered for %d commands + ^axs: callbacks", len(CMDS))
